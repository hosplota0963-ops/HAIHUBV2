local function SaveConfig()
    pcall(function()
        writefile(ConfigName, HttpService:JSONEncode(Config))
    end)
end

local function LoadConfig()
    pcall(function()
        if isfile(ConfigName) then
            local data = HttpService:JSONDecode(readfile(ConfigName))
            for k, v in pairs(data) do Config[k] = v end
        end
    end)
end

local FlyConnection
local IsFlying = false
local FlyBodyVelocity = nil
local FlyBodyGyro = nil

PlayerTab:New("Toggle")({
    Title = "Fly (บินแก้บัค 100%)",
    Description = "เปิด/ปิด การบิน (ใช้ W A S D บังคับ)",
    DefaultValue = false,
    Callback = function(Value)
        IsFlying = Value
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChild("Humanoid")
        
        if hrp and hum then
            if IsFlying then
                FlyBodyVelocity = Instance.new("BodyVelocity")
                FlyBodyVelocity.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                FlyBodyVelocity.Velocity = Vector3.zero
                FlyBodyVelocity.Parent = hrp
                
                FlyBodyGyro = Instance.new("BodyGyro")
                FlyBodyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
                FlyBodyGyro.P = 10000
                FlyBodyGyro.D = 50
                FlyBodyGyro.Parent = hrp
                
                FlyConnection = RunService.RenderStepped:Connect(function()
                    if not LocalPlayer.Character or not LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then return end
                    local cam = workspace.CurrentCamera
                    local moveDir = Vector3.new(0, 0, 0)
                    
                    if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + cam.CFrame.LookVector end
                    if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - cam.CFrame.LookVector end
                    if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - cam.CFrame.RightVector end
                    if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + cam.CFrame.RightVector end
                    
                    FlyBodyVelocity.Velocity = moveDir * Config.FlySpeed
                    FlyBodyGyro.CFrame = cam.CFrame
                    LocalPlayer.Character.Humanoid.PlatformStand = true
                end)
            else
                if FlyBodyVelocity then FlyBodyVelocity:Destroy() end
                if FlyBodyGyro then FlyBodyGyro:Destroy() end
                if FlyConnection then FlyConnection:Disconnect() end
                hum.PlatformStand = false
            end
        end
    end,
})

RunService.Stepped:Connect(function()
    -- ถ้าหน้าจอเลือกทีมเปิดอยู่ ห้ามขยับตัว หรือรันฟังก์ชันช่วยเล่นทุกชนิด
    if isSelectingTeam() then return end 

    local char = LocalPlayer.Character
    if not char then return end

    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChild("Humanoid")

    -- รวมระบบ NoClip / SafeMode เข้าด้วยกัน
    local isNoClipping = Config.NoClip or Config.SafeMode or Config.AutoATM or Config.AutoFixCar
    if isNoClipping then
        for _, part in pairs(char:GetDescendants()) do
            if part:IsA("BasePart") and part.CanCollide then
                -- 🔥 ป้องกันบัค: ถ้าเปิดตีกระเด็น ห้ามทะลุส่วนลำตัว ไม่งั้นจะตีไม่โดนคนอื่น!!
                if Config.GodWeaponFling and (part.Name == "HumanoidRootPart" or part.Name == "Torso" or string.match(part.Name, "Torso")) then
                    continue
                end
                part.CanCollide = false
            end
        end
    end

    -- ระบบ Safe Mode (ทะลุกำแพงแต่ไม่ตกพื้น)
    if Config.SafeMode and hrp and hum then
        local params = RaycastParams.new()
        params.FilterType = RaycastFilterType.Exclude
        params.FilterDescendantsInstances = {char}
        
        local result = workspace:Raycast(hrp.Position, Vector3.new(0, -8, 0), params)
        if result then
            local floorY = result.Position.Y + (hum.HipHeight or 2) + (hrp.Size.Y / 2)
            -- ใส่ Threshold (-1) ป้องกันสคริปต์สแปมดึงตัวขึ้นลงรัวๆ จนขัดจังหวะฟิสิกส์ตีกระเด็น
            if hrp.Position.Y < floorY - 1 then
                hrp.CFrame = CFrame.new(hrp.Position.X, floorY + 0.1, hrp.Position.Z) * hrp.CFrame.Rotation
                hrp.Velocity = Vector3.new(hrp.Velocity.X, 0, hrp.Velocity.Z)
            end
        end
    end

    -- ระบบ Anti Fall (กันตกแมพ)
    if Config.AntiFall and hrp and hum then
        local destroyHeight = workspace.FallenPartsDestroyHeight or -500
        -- ถ้าหล่นลงไปใกล้ขอบมรณะ จะงัดกลับมาที่เดิม
        if hrp.Position.Y < destroyHeight + 100 then
            if LastSafePosition then
                hrp.CFrame = CFrame.new(LastSafePosition)
                hrp.Velocity = Vector3.zero
                hrp.RotVelocity = Vector3.zero
            end
        else
            -- อัปเดตจุดปลอดภัยล่าสุดตลอดเวลาที่ยืนปลอดภัย
            if hum.FloorMaterial ~= Enum.Material.Air then
                LastSafePosition = hrp.Position + Vector3.new(0, 3, 0)
            end
        end
    end
    
    -- ระบบวิ่งไว (Bypass แบบบังคับความเร็วผ่าน Velocity)
    if Config.EnableWalkSpeed and hrp and hum then
        if hum.MoveDirection.Magnitude > 0 then
            hrp.Velocity = Vector3.new(hum.MoveDirection.X * Config.WalkSpeed, hrp.Velocity.Y, hum.MoveDirection.Z * Config.WalkSpeed)
        end
    end

    if Config.LockPlayer and SelectedPlayer ~= "" then
        pcall(function()
            local target = Players:FindFirstChild(SelectedPlayer)
            if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") and hrp then
                target.Character.HumanoidRootPart.CFrame = hrp.CFrame * CFrame.new(0, 0, -3)
                target.Character.HumanoidRootPart.Velocity = Vector3.zero
                target.Character.HumanoidRootPart.RotVelocity = Vector3.zero
            end
        end)
    end

    if Config.AutoKillPlayer and SelectedPlayer ~= "" then
        pcall(function()
            local target = Players:FindFirstChild(SelectedPlayer)
            if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") and hrp then
                local targetHrp = target.Character.HumanoidRootPart
                local targetPos = targetHrp.Position
                local underPos = targetPos - Vector3.new(0, 4, 0)
                hrp.CFrame = CFrame.lookAt(underPos, targetPos)
                hrp.Velocity = Vector3.zero 
                hrp.RotVelocity = Vector3.zero
            end
        end)
    end

    if Config.BigWeapon then
        pcall(function()
            local function resizeWeapon(parentFolder)
                local weapon = parentFolder:FindFirstChild("steell bat")
                if weapon and weapon:FindFirstChild("Handle") then
                    weapon.Handle.Size = Vector3.new(50, 50, 50)
                    weapon.Handle.Transparency = 0.5
                    weapon.Handle.Massless = true
                    weapon.Handle.CanCollide = false
                end
            end
            resizeWeapon(char)
            if LocalPlayer:FindFirstChild("Backpack") then resizeWeapon(LocalPlayer.Backpack) end
        end)
    end
    
    -- ระบบ God Fling (เวอร์ชันแรกสุด ดั้งเดิม ปั่น 50,000 แกน Y แต่เพิ่มลบแรงเสียดทานให้เดินได้)
    if Config.GodWeaponFling and hrp then
        pcall(function()
            -- ลบแรงเสียดทาน (Friction) ให้เราสามารถเดินไปมาได้ แม้ตัวจะหมุนอยู่
            hrp.CustomPhysicalProperties = PhysicalProperties.new(100, 0, 0, 0, 0)
            
            -- ปั่นแกน Y ด้วยความเร็ว 50,000 (เหมือนเวอร์ชันแรกที่ตีกระเด็นได้จริง)
            hrp.RotVelocity = Vector3.new(0, 50000, 0)
            
            -- ขยายอาวุธให้ตีโดนง่ายขึ้น
            local function resizeAllWeapons(parentFolder)
                for _, tool in pairs(parentFolder:GetChildren()) do
                    if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
                        tool.Handle.Size = Vector3.new(60, 60, 60)
                        tool.Handle.Transparency = 1 
                        tool.Handle.Massless = true
                        tool.Handle.CanCollide = false
                    end
                end
            end
            resizeAllWeapons(char)
            if LocalPlayer:FindFirstChild("Backpack") then resizeAllWeapons(LocalPlayer.Backpack) end
        end)
    end
end)

local ESPObjects = {}

local function CreateESP(player)
    if player == LocalPlayer then return end
    
    local esp = {
        Tracer = Drawing.new("Line"),
        Text = Drawing.new("Text"),
        Highlight = Instance.new("Highlight")
    }
    
    esp.Tracer.Visible = false
    esp.Tracer.Color = Color3.fromRGB(0, 255, 255)
    esp.Tracer.Thickness = 1
    esp.Tracer.Transparency = 1
    
    esp.Text.Visible = false
    esp.Text.Center = true
    esp.Text.Outline = true
    esp.Text.Color = Color3.fromRGB(255, 255, 255)
    esp.Text.Size = 16
    
    esp.Highlight.FillColor = Color3.fromRGB(255, 0, 0)
    esp.Highlight.FillTransparency = 0.8
    esp.Highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
    esp.Highlight.OutlineTransparency = 0.2
    esp.Highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    esp.Highlight.Parent = CoreGui
    
    ESPObjects[player] = esp
end

local function RemoveESP(player)
    if ESPObjects[player] then
        ESPObjects[player].Tracer:Remove()
        ESPObjects[player].Text:Remove()
        if ESPObjects[player].Highlight then ESPObjects[player].Highlight:Destroy() end
        ESPObjects[player] = nil
    end
end

                esp.Highlight.Enabled = false
            end
            
            if onScreen then
                if Config.ESP_Tracer then
                    esp.Tracer.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                    esp.Tracer.To = Vector2.new(vector.X, vector.Y)
                    esp.Tracer.Visible = true
                else
                    esp.Tracer.Visible = false
                end
                
                if Config.ESP_NameDist then
                    esp.Text.Position = Vector2.new(headVector.X, headVector.Y - 20)
                    esp.Text.Text = string.format("%s [%d m]", player.Name, distance)
                    esp.Text.Visible = true
                else
                    esp.Text.Visible = false
                end
            else
                esp.Tracer.Visible = false
                esp.Text.Visible = false
            end
        else
            esp.Tracer.Visible = false
            esp.Text.Visible = false
            esp.Highlight.Enabled = false
        end
    end
end)
